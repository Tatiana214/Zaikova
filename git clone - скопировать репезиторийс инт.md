git clone - скопировать репезиторийс интернетак себе
git status - проверить что изменилось
git add - добавить файл, что бы Git начал занимследить
git commit - сохранить изменения с комментарием
git push - отправить свои сохранения на сервер
git init - создать новый репозистор в папке
git log - посмотреть историю всех сохранений
git diff - показать, что изменилось в файлах 
git branch - посмотреть списокветок
git checkout - переключитьсяна другую ветку
